package com.example.simplealarm;

import androidx.appcompat.app.AppCompatActivity;

import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.Toast;

import android.app.AlarmManager;
import android.app.PendingIntent;
import android.content.Context;
import android.content.Intent;
import android.os.Build;
import android.content.pm.PackageManager;

public class MainActivity extends AppCompatActivity {

    Button btnSetAlarm, btnCancelAlarm;
    PendingIntent pendingIntent;
    AlarmManager alarmManager;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        btnSetAlarm = (Button) findViewById(R.id.btnSetAlarm);
        btnCancelAlarm = (Button) findViewById(R.id.btnCancelAlarm);

        alarmManager = (AlarmManager) getSystemService(Context.ALARM_SERVICE);

        final Intent intent = new Intent(this, AlarmReceiver.class);

        pendingIntent = PendingIntent.getBroadcast(this, 0, intent, PendingIntent.FLAG_IMMUTABLE);

        // ----------------------------
        // 1️⃣ Check exact alarm permission (Android 12+)
        // ----------------------------
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) { // Android 12+
            if (!alarmManager.canScheduleExactAlarms()) {
                Intent intentSettings = new Intent(android.provider.Settings.ACTION_REQUEST_SCHEDULE_EXACT_ALARM);
                startActivity(intentSettings);
                Toast.makeText(this, "Please allow exact alarms for this app", Toast.LENGTH_LONG).show();
            }
        }

        // ----------------------------
        // 2️⃣ Request notification permission (Android 13+)
        // ----------------------------
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            if (checkSelfPermission(android.Manifest.permission.POST_NOTIFICATIONS)
                    != PackageManager.PERMISSION_GRANTED) {

                requestPermissions(
                        new String[]{android.Manifest.permission.POST_NOTIFICATIONS},
                        100
                );
            }
        }

        // ----------------------------
        // 3️⃣ Button to set alarm
        // ----------------------------
        btnSetAlarm.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View view) {
                setAlarm();
            }
        });

        // ----------------------------
        // 4️⃣ Button to cancel alarm
        // ----------------------------
        btnCancelAlarm.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View view) {
                cancelAlarm();
            }
        });
    }

    // ----------------------------
    // Handle notification permission result
    // ----------------------------
    @Override
    public void onRequestPermissionsResult(int requestCode, String[] permissions, int[] grantResults) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults);

        if (requestCode == 100) {
            if (grantResults.length > 0 && grantResults[0] == PackageManager.PERMISSION_GRANTED) {
                Toast.makeText(this, "Notifications permission granted", Toast.LENGTH_SHORT).show();
            } else {
                Toast.makeText(this, "Permission denied. Notifications may be blocked!", Toast.LENGTH_LONG).show();
            }
        }
    }

    // ----------------------------
    // Set alarm for 10 seconds later
    // ----------------------------
    private void setAlarm() {
        long triggerTime = System.currentTimeMillis() + 10 * 1000; // 10 seconds

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            alarmManager.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, triggerTime, pendingIntent);
        } else {
            alarmManager.setExact(AlarmManager.RTC_WAKEUP, triggerTime, pendingIntent);
        }

        Toast.makeText(MainActivity.this, "Alarm set for 10 seconds", Toast.LENGTH_SHORT).show();
    }

    // ----------------------------
    // Cancel alarm
    // ----------------------------
    private void cancelAlarm() {
        alarmManager.cancel(pendingIntent);
        Toast.makeText(MainActivity.this, "Alarm canceled", Toast.LENGTH_SHORT).show();
    }
}
