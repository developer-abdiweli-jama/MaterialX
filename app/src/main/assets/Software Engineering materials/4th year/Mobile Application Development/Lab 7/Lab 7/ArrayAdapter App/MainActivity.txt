package com.example.arrayadapterapp;

import android.os.Bundle;
import android.view.View;
import android.widget.AdapterView;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.EditText;
import android.widget.ListView;
import android.widget.TextView;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import java.util.ArrayList;
import java.util.List;

public class MainActivity extends AppCompatActivity {

    // Declare variables
    private ListView listView;
    private EditText inputEditText;
    private Button addButton, clearButton, countButton;

    private List<String> itemList;
    private ArrayAdapter<String> adapter;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        // Initialize views
        initializeViews();

        // Initialize data
        initializeData();

        // Initialize adapter
        initializeAdapter();

        // Set up click listeners
        setupClickListeners();
    }

    private void initializeViews() {
        listView = findViewById(R.id.listView);
        inputEditText = findViewById(R.id.inputEditText);
        addButton = findViewById(R.id.addButton);
        clearButton = findViewById(R.id.clearButton);
        countButton = findViewById(R.id.countButton);
    }

    private void initializeData() {
        // Create initial data
        itemList = new ArrayList<>();
        itemList.add("Apple");
        itemList.add("Banana");
        itemList.add("Orange");
        itemList.add("Grapes");
        itemList.add("Mango");
    }

    private void initializeAdapter() {
        // Create ArrayAdapter with built-in layout
        adapter = new ArrayAdapter<>(
                this,                                   // Context
                android.R.layout.simple_list_item_1,    // Built-in layout
                itemList                                // Data
        );

        // Alternative: Using custom layout
        /*
        adapter = new ArrayAdapter<>(
            this,
            R.layout.custom_list_item,              // Custom layout
            android.R.id.text1,                     // TextView ID in layout
            itemList                                // Data
        );
        */

        // Set adapter to ListView
        listView.setAdapter(adapter);
    }

    private void setupClickListeners() {
        // Add button click
        addButton.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                addNewItem();
            }
        });

        // List item click
        listView.setOnItemClickListener(new AdapterView.OnItemClickListener() {
            @Override
            public void onItemClick(AdapterView<?> parent, View view, int position, long id) {
                String selectedItem = itemList.get(position);
                Toast.makeText(MainActivity.this,
                        "Clicked: " + selectedItem + " (Position: " + position + ")",
                        Toast.LENGTH_SHORT).show();
            }
        });

        // List item long click
        listView.setOnItemLongClickListener(new AdapterView.OnItemLongClickListener() {
            @Override
            public boolean onItemLongClick(AdapterView<?> parent, View view, int position, long id) {
                String removedItem = itemList.get(position);
                itemList.remove(position);
                adapter.notifyDataSetChanged(); // Refresh the list

                Toast.makeText(MainActivity.this,
                        "Removed: " + removedItem,
                        Toast.LENGTH_SHORT).show();
                return true; // Consume the event
            }
        });

        // Clear all button
        clearButton.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                clearAllItems();
            }
        });

        // Count button
        countButton.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                showItemCount();
            }
        });
    }

    private void addNewItem() {
        String newItem = inputEditText.getText().toString().trim();

        if (newItem.isEmpty()) {
            Toast.makeText(this, "Please enter an item", Toast.LENGTH_SHORT).show();
            return;
        }

        // Add to list
        itemList.add(newItem);

        // Notify adapter that data changed
        adapter.notifyDataSetChanged();

        // Clear input field
        inputEditText.setText("");

        // Scroll to the new item
        listView.smoothScrollToPosition(itemList.size() - 1);

        Toast.makeText(this, "Added: " + newItem, Toast.LENGTH_SHORT).show();
    }

    private void clearAllItems() {
        if (itemList.isEmpty()) {
            Toast.makeText(this, "List is already empty", Toast.LENGTH_SHORT).show();
            return;
        }

        itemList.clear();
        adapter.notifyDataSetChanged();
        Toast.makeText(this, "All items cleared", Toast.LENGTH_SHORT).show();
    }

    private void showItemCount() {
        int count = itemList.size();
        Toast.makeText(this,
                "Total items: " + count,
                Toast.LENGTH_LONG).show();
    }

    // Optional: Handle Enter key in EditText
    private void setupInputListeners() {
        inputEditText.setOnKeyListener((v, keyCode, event) -> {
            if (keyCode == android.view.KeyEvent.KEYCODE_ENTER) {
                addNewItem();
                return true;
            }
            return false;
        });
    }
}
